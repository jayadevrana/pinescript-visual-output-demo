//@version=6
indicator("Lesson 4 Visual Output Demo", overlay=true)

length = input.int(8, "MA Length", minval=1)
ma = ta.ema(close, length)

upperZone = ma + 800
lowerZone = ma - 800

buySignal = ta.crossover(close, ma)
sellSignal = ta.crossunder(close, ma)
aboveMa = close > ma
trendUp = ta.rising(ma, 3)

maPlot = plot(ma, title="Moving Average", color=color.aqua, linewidth=2)
upperPlot = plot(upperZone, title="Upper Fill Guide", color=color.new(color.aqua, 100), display=display.none)
lowerPlot = plot(lowerZone, title="Lower Fill Guide", color=color.new(color.aqua, 100), display=display.none)

supportLine = hline(65000, "Support", color=color.new(color.green, 0), linewidth=2)
resistanceLine = hline(70000, "Resistance", color=color.new(color.red, 0), linewidth=2)

fill(upperPlot, lowerPlot, title="Average Zone", color=color.new(color.aqua, 80))
fill(supportLine, resistanceLine, title="Price Zone", color=color.new(color.orange, 86))

plotshape(buySignal, title="Buy Label", style=shape.labelup, location=location.belowbar, color=color.lime, text="Buy", textcolor=color.white, size=size.small)
plotshape(sellSignal, title="Sell Label", style=shape.labeldown, location=location.abovebar, color=color.red, text="Sell", textcolor=color.white, size=size.small)
plotshape(buySignal, title="Triangle Up", style=shape.triangleup, location=location.belowbar, color=color.green, offset=1, size=size.small)
plotshape(sellSignal, title="Triangle Down", style=shape.triangledown, location=location.abovebar, color=color.maroon, offset=1, size=size.small)
plotshape(trendUp, title="Circle Marker", style=shape.circle, location=location.belowbar, color=color.orange, offset=-1, size=size.tiny)

plotchar(aboveMa, title="Above Average Character", char="A", location=location.top, color=color.black)

bgcolor(aboveMa ? color.new(color.teal, 85) : color.new(color.orange, 85), title="Trend Background")
barcolor(aboveMa ? color.blue : color.fuchsia, title="Trend Candle Color")
